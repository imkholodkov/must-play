32def transform_row(row: dict[str, str]) -> GameRecord: 

33 """Очистить строку CSV. Бросает RejectedRow, если строку нельзя загрузить.""" 

34 app_id = cleaning.parse_int(row.get("app_id")) 

35 if app_id is None or app_id <= 0: 

36 raise RejectedRow("bad_app_id") 

37 name = cleaning.normalize_name(row.get("name") or "") 

38 if not name: 

39 raise RejectedRow("empty_name") 

40 

41 positive = cleaning.parse_int(row.get("positive")) or 0 

42 negative = cleaning.parse_int(row.get("negative")) or 0 

43 if positive < 0 or negative < 0: 

44 raise RejectedRow("negative_votes") 

45 

46 owners = cleaning.parse_owners(row.get("estimated_owners")) 

47 price = cleaning.parse_price(row.get("price")) 

48 game = { 

49 "app_id": app_id, 

50 "name": name[:500], 

51 "release_date": cleaning.parse_date(row.get("release_date")), 

52 "price": price, 

53 "is_free": row.get("price_status") == "free" or price == 0, 

54 "owners_min": owners[0] if owners else None, 

55 "owners_max": owners[1] if owners else None, 

56 "positive": positive, 

57 "negative": negative, 

58 "recommendations": cleaning.parse_int(row.get("recommendations")), 

59 "peak_ccu": cleaning.parse_int(row.get("peak_ccu")), 

60 "metacritic_score": cleaning.nullify_zero_score(row.get("metacritic_score")), 

61 "achievements": cleaning.parse_int(row.get("achievements")), 

62 "dlc_count": cleaning.parse_int(row.get("dlc_count")), 

63 "windows": cleaning.parse_bool(row.get("windows")), 

64 "mac": cleaning.parse_bool(row.get("mac")), 

65 "linux": cleaning.parse_bool(row.get("linux")), 

66 "short_description": (row.get("short_description") or "").strip() or None, 

67 "header_image": (row.get("header_image") or "").strip()[:500] or None, 

68 } 

69 lists = {key: parse(row.get(column)) for key, (column, parse) in LIST_FIELDS.items()} 

70 return GameRecord(game=game, lists=lists)
