12Z_95 = 1.96 

13MIN_VOTES = 50 # порог из лабы 6: при меньшем числе голосов доля положительных слишком шумная

28def wilson_lower_bound(positive: int, negative: int, z: float = Z_95) -> float | None: 

29 """Нижняя граница доверительного интервала Уилсона для доли положительных, 0–100. 

30 

31 3 голоса «за» из 3 дают около 44, а 9 000 из 10 000 — около 89: мало голосов не выводит игру наверх. 

32 """ 

33 n = positive + negative 

34 if n == 0: 

35 return None 

36 p = positive / n 

37 centre = p + z * z / (2 * n) 

38 margin = z * math.sqrt((p * (1 - p) + z * z / (4 * n)) / n) 

39 return round((centre - margin) / (1 + z * z / n) * 100, 2)

51def score_game(positive: int, negative: int, critic_avg: float | None, critic_count: int) -> Score: 

52 """Посчитать оценки одной игры по правилам 1–3. 

53 

54 critic_score сохраняется как есть (средняя оценка прессы), а в индекс и сравнение 

55 с игроками идёт оценка с поправкой на щедрость прессы. 

56 """ 

57 player = wilson_lower_bound(positive, negative) 

58 critic = round(critic_avg, 2) if critic_avg is not None and critic_count >= MIN_CRITIC_REVIEWS else None 

59 adjusted = max(critic - CRITIC_OFFSET, 0.0) if critic is not None else None 

60 eligible = positive + negative >= MIN_VOTES 

61 

62 if player is None: 

63 index = None 

64 elif adjusted is None: 

65 index = player 

66 else: 

67 index = round(CRITIC_WEIGHT * adjusted + PLAYER_WEIGHT * player, 2) 

68 

69 divergence = None 

70 share = positive / (positive + negative) * 100 if positive + negative else None 

71 if adjusted is not None and share is not None and eligible and abs(adjusted - share) > DIVERGENCE_POINTS: 

72 divergence = "overrated" if adjusted > share else "underrated" 

73 return Score(player, critic, index, eligible, divergence)

76def next_must_play_status(current: str | None, index: float | None, eligible: bool) -> str | None: 

77 """Правило 4: candidate → in_list → dropped. 

78 

79 - игра впервые выше порога входа — кандидат; 

80 - кандидат, удержавшийся выше порога выхода в следующем пересчёте, — в списке; 

81 - игра из списка (или кандидат) ниже порога выхода — выбыла; 

82 - выбывшая игра снова выше порога входа — снова кандидат. 

83 Возвращает None, если игра в must play не участвует. 

84 """ 

85 above_in = eligible and index is not None and index >= MUST_PLAY_IN 

86 above_out = eligible and index is not None and index >= MUST_PLAY_OUT 

87 if current in (None, DROPPED): 

88 return CANDIDATE if above_in else current 

89 if current == CANDIDATE: 

90 return IN_LIST if above_out else DROPPED 

91 return IN_LIST if above_out else DROPPED
