21class RatingInProgressError(Exception): 

22 """Другой пересчёт ещё идёт — второй запускать нельзя (API ответит 409)."""

25def _chunks(rows: list[dict]): 

26 for start in range(0, len(rows), BATCH_SIZE): 

27 yield rows[start : start + BATCH_SIZE]

30def recalculate(session: Session) -> RatingRun: 

31 """Пересчитать рейтинг всех игр в одной транзакции и вернуть запись о запуске.""" 

32 if session.scalar(select(RatingRun.id).where(RatingRun.status == "running").with_for_update()): 

33 raise RatingInProgressError("пересчёт рейтинга уже выполняется") 

34 run = RatingRun(status="running") 

35 session.add(run) 

36 session.commit() 

37 

38 try: 

39 critic = dict( 

40 (app_id, (avg, cnt)) 

41 for app_id, avg, cnt in session.execute( 

42 select(CriticReview.app_id, func.avg(CriticReview.score), func.count(CriticReview.score)) 

43 .where(CriticReview.score.is_not(None)) 

44 .group_by(CriticReview.app_id) 

45 ) 

46 ) 

47 statuses = dict(session.execute(select(MustPlayEntry.app_id, MustPlayEntry.status)).tuples().all()) 

48 

49 scores, history, must_play = [], [], [] 

50 for app_id, positive, negative in session.execute(select(Game.app_id, Game.positive, Game.negative)): 

51 avg, cnt = critic.get(app_id, (None, 0)) 

52 s = score_game(positive, negative, float(avg) if avg is not None else None, cnt) 

53 scores.append( 

54 { 

55 "app_id": app_id, 

56 "run_id": run.id, 

57 "player_score": s.player_score, 

58 "critic_score": s.critic_score, 

59 "critic_reviews_count": cnt, 

60 "index_score": s.index_score, 

61 "is_eligible": s.is_eligible, 

62 "divergence": s.divergence, 

63 } 

64 ) 

65 if s.is_eligible and s.index_score is not None: 

66 history.append({"run_id": run.id, "app_id": app_id, "index_score": s.index_score}) 

67 current = statuses.get(app_id) 

68 new = next_must_play_status(current, s.index_score, s.is_eligible) 

69 if new is not None and new != current: 

70 must_play.append({"app_id": app_id, "status": new, "run_id": run.id}) 

71 

72 for rows in _chunks(scores): 

73 stmt = insert(GameScore).values(rows) 

74 cols = {c: stmt.excluded[c] for c in rows[0] if c != "app_id"} 

75 session.execute(stmt.on_conflict_do_update(index_elements=["app_id"], set_=cols)) 

76 for rows in _chunks(history): 

77 session.execute(insert(ScoreHistory), rows) 

78 for rows in _chunks(must_play): 

79 stmt = insert(MustPlayEntry).values(rows) 

80 session.execute( 

81 stmt.on_conflict_do_update( 

82 index_elements=["app_id"], 

83 set_={"status": stmt.excluded.status, "run_id": stmt.excluded.run_id, "changed_at": func.now()}, 

84 ) 

85 ) 

86 

87 run.games_scored = len(scores) 

88 run.status = "success" 

89 run.finished_at = datetime.now(UTC) 

90 session.commit() 

91 except BaseException: 

92 session.rollback() 

93 run = session.merge(run) 

94 run.status = "failed" 

95 run.finished_at = datetime.now(UTC) 

96 session.commit() 

97 raise 

98 return run
