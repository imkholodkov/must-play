38def _features(session: Session, app_ids: list[int]) -> dict[int, list[str]]: 

39 """Признаки игры как список токенов: g:Action, t:Roguelike, c:Co-op.""" 

40 tokens: dict[int, list[str]] = defaultdict(list) 

41 for prefix, link, model, column in ( 

42 ("g", game_genres, Genre, game_genres.c.genre_id), 

43 ("t", game_tags, Tag, game_tags.c.tag_id), 

44 ("c", game_categories, Category, game_categories.c.category_id), 

45 ): 

46 rows = session.execute( 

47 select(link.c.app_id, model.name).join(model, model.id == column).where(link.c.app_id.in_(app_ids)) 

48 ) 

49 for app_id, name in rows: 

50 tokens[app_id].append(f"{prefix}:{name}") 

51 return tokens

54def build_similarities(session: Session, top_k: int = TOP_K) -> dict: 

55 """Пересчитать таблицу game_similarities для допущенных игр в одной транзакции.""" 

56 games = dict( 

57 session.execute( 

58 select(GameScore.app_id, GameScore.index_score).where( 

59 GameScore.is_eligible, GameScore.index_score.is_not(None) 

60 ) 

61 ).all() 

62 ) 

63 tokens = _features(session, list(games)) 

64 app_ids = [a for a in games if tokens.get(a)] 

65 if len(app_ids) < 2: 

66 return {"games": 0, "pairs": 0, "genre_precision": None} 

67 

68 vectorizer = TfidfVectorizer(analyzer=lambda doc: doc) # документ уже разбит на токены 

69 matrix = vectorizer.fit_transform([tokens[a] for a in app_ids]) 

70 neighbours = NearestNeighbors(n_neighbors=min(CANDIDATES + 1, len(app_ids)), metric="cosine").fit(matrix) 

71 distances, indices = neighbours.kneighbors(matrix) 

72 

73 rows, hits, total = [], 0, 0 

74 for row, app_id in enumerate(app_ids): 

75 genres = {t for t in tokens[app_id] if t.startswith("g:")} 

76 scored = [] 

77 for dist, idx in zip(distances[row], indices[row], strict=True): 

78 other = app_ids[idx] 

79 if other == app_id: 

80 continue 

81 similarity = 1 - dist 

82 rank_score = SIMILARITY_WEIGHT * similarity + QUALITY_WEIGHT * games[other] / 100 

83 scored.append((rank_score, other, similarity)) 

84 scored.sort(reverse=True) 

85 for rank, (_, other, similarity) in enumerate(scored[:top_k], start=1): 

86 rows.append({"app_id": app_id, "similar_app_id": other, "rank": rank, "score": round(similarity, 4)}) 

87 total += 1 

88 hits += bool(genres & {t for t in tokens[other] if t.startswith("g:")}) 

89 

90 session.execute(delete(GameSimilarity)) 

91 for start in range(0, len(rows), 10000): 

92 session.execute(insert(GameSimilarity), rows[start : start + 10000]) 

93 session.commit() 

94 # Доля рекомендаций, у которых есть общий жанр с исходной игрой (precision@K по жанру) 

95 return {"games": len(app_ids), "pairs": len(rows), "genre_precision": round(hits / total, 3)}

98def similar_games(session: Session, app_id: int, limit: int) -> list[tuple[Game, float]]: 

99 rows = session.execute( 

100 select(Game, GameSimilarity.score) 

101 .join(GameSimilarity, GameSimilarity.similar_app_id == Game.app_id) 

102 .where(GameSimilarity.app_id == app_id) 

103 .options(selectinload(Game.genres), selectinload(Game.score)) 

104 .order_by(GameSimilarity.rank) 

105 .limit(limit) 

106 ) 

107 return [(game, score) for game, score in rows]

121def recommend(session: Session, prefs: Preferences, limit: int) -> list[Game]: 

122 """Подбор по анкете: допущенные игры, подходящие под все условия, по убыванию индекса.""" 

123 stmt = select(Game).join(GameScore, GameScore.app_id == Game.app_id).where(GameScore.is_eligible) 

124 for name in prefs.genres: 

125 stmt = stmt.where(Game.genres.any(func.lower(Genre.name) == name.lower())) 

126 for name in prefs.tags: 

127 stmt = stmt.where(Game.tags.any(func.lower(Tag.name) == name.lower())) 

128 if prefs.mode: 

129 stmt = stmt.where(Game.categories.any(Category.name.in_(MODES[prefs.mode]))) 

130 if prefs.platform: 

131 stmt = stmt.where(getattr(Game, prefs.platform).is_(True)) 

132 if prefs.language: 

133 stmt = stmt.where(Game.languages.any(func.lower(Language.name) == prefs.language.lower())) 

134 if prefs.max_price is not None: 

135 stmt = stmt.where((Game.price <= prefs.max_price) | Game.is_free) 

136 if prefs.exclude: 

137 stmt = stmt.where(Game.app_id.not_in(prefs.exclude)) 

138 stmt = stmt.options(selectinload(Game.genres), selectinload(Game.score)) 

139 return list(session.scalars(stmt.order_by(GameScore.index_score.desc(), Game.app_id).limit(limit)))
