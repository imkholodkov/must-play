1"""Загрузка очищенных записей в PostgreSQL пачками, без дублей при повторном запуске.""" 

2 

3from sqlalchemy import Table, delete, select 

4from sqlalchemy.dialects.postgresql import insert 

5from sqlalchemy.orm import Session 

6 

7from app.etl.transform import GameRecord 

8from app.models import Category, Developer, Game, Genre, Language, Tag 

9from app.models.catalog import game_categories, game_developers, game_genres, game_languages, game_tags 

10 

11# Справочник → (модель, таблица связи, имя колонки справочника в таблице связи) 

12DICTIONARIES = { 

13 "genres": (Genre, game_genres, "genre_id"), 

14 "tags": (Tag, game_tags, "tag_id"), 

15 "categories": (Category, game_categories, "category_id"), 

16 "languages": (Language, game_languages, "language_id"), 

17 "developers": (Developer, game_developers, "developer_id"), 

18} 

19NAME_LIMITS = {"genres": 100, "tags": 100, "categories": 150, "languages": 100, "developers": 300} 

20 

21 

22class Loader: 

23 """Держит кэш справочников, чтобы не искать одно и то же имя в БД на каждой пачке.""" 

24 

25 def __init__(self, session: Session): 

26 self.session = session 

27 self.ids: dict[str, dict[str, int]] = {key: {} for key in DICTIONARIES} 

28 

29 def load_batch(self, records: list[GameRecord]) -> tuple[int, int]: 

30 """Записать пачку игр со связями. Возвращает (новых, обновлённых).""" 

31 app_ids = [r.game["app_id"] for r in records] 

32 existing = set(self.session.scalars(select(Game.app_id).where(Game.app_id.in_(app_ids)))) 

33 

34 stmt = insert(Game).values([r.game for r in records]) 

35 update_cols = {c: stmt.excluded[c] for c in records[0].game if c != "app_id"} 

36 self.session.execute(stmt.on_conflict_do_update(index_elements=["app_id"], set_=update_cols)) 

37 

38 for key, (model, link, column) in DICTIONARIES.items(): 

39 self._replace_links(key, model, link, column, records, app_ids) 

40 return len(app_ids) - len(existing), len(existing) 

41 

42 def _replace_links(self, key, model, link: Table, column: str, records, app_ids) -> None: 

43 limit = NAME_LIMITS[key] 

44 names = {name[:limit] for r in records for name in r.lists[key]} 

45 ids = self._ensure_names(key, model, names) 

46 # Связи игры перезаписываются целиком: повторный импорт не плодит дубли и убирает устаревшие 

47 self.session.execute(delete(link).where(link.c.app_id.in_(app_ids))) 

48 rows = {(r.game["app_id"], ids[name[:limit]]) for r in records for name in r.lists[key]} 

49 if rows: 

50 self.session.execute(link.insert(), [{"app_id": a, column: i} for a, i in rows]) 

51 

52 def _ensure_names(self, key, model, names: set[str]) -> dict[str, int]: 

53 cache = self.ids[key] 

54 missing = names - cache.keys() 

55 if missing: 

56 self.session.execute( 

57 insert(model).values([{"name": n} for n in missing]).on_conflict_do_nothing(index_elements=["name"]) 

58 ) 

59 for name, id_ in self.session.execute(select(model.name, model.id).where(model.name.in_(missing))): 

60 cache[name] = id_ 

61 return cache
