28def read_rows(path: Path) -> Iterator[dict[str, str]]: 

29 """Читать CSV построчно: файл на 925 МБ целиком в память не грузится.""" 

30 with path.open(encoding="utf-8", newline="") as f: 

31 yield from csv.DictReader(f)

34def run_import(session: Session, source: Path, batch_size: int = 1000) -> EtlRun: 

35 """Импортировать каталог в одной транзакции и записать отчёт в etl_runs.""" 

36 run = EtlRun(source=str(source), status="running", rejection_reasons={}) 

37 session.add(run) 

38 session.commit() 

39 

40 reasons: Counter[str] = Counter() 

41 seen: set[int] = set() 

42 batch: list[GameRecord] = [] 

43 loader = Loader(session) 

44 

45 def flush() -> None: 

46 loaded, updated = loader.load_batch(batch) 

47 run.rows_loaded += loaded 

48 run.rows_updated += updated 

49 batch.clear() 

50 

51 try: 

52 for row in read_rows(source): 

53 run.rows_read += 1 

54 try: 

55 record = transform_row(row) 

56 if record.game["app_id"] in seen: 

57 raise RejectedRow("duplicate_app_id") 

58 except RejectedRow as exc: 

59 reasons[exc.reason] += 1 

60 continue 

61 seen.add(record.game["app_id"]) 

62 batch.append(record) 

63 if len(batch) >= batch_size: 

64 flush() 

65 logger.info("Обработано строк: %s", run.rows_read) 

66 if batch: 

67 flush() 

68 run.status = "success" 

69 except BaseException: 

70 # BaseException: прерывание по Ctrl+C тоже должно пометить запуск как failed, а не оставить running 

71 session.rollback() 

72 run = session.merge(run) 

73 run.status = "failed" 

74 raise 

75 finally: 

76 run.rows_rejected = sum(reasons.values()) 

77 run.rejection_reasons = dict(reasons) 

78 run.finished_at = datetime.now(UTC) 

79 session.commit() 

80 return run
